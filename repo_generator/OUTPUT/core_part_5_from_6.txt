# ============================================================================
#  Proper File Protocol (PFP) v4.1.3 STABLE
#  Project: HQ2 Data Ingestion & Normalization Stack
#  Layer: DETECTOR LAYER (SEMANTIC DECISION ENGINE)
#  File: data_preprocessing/core/date_detector/date_detector.py
# ----------------------------------------------------------------------------
#  ROLE
#    Deterministic semantic decision layer.
#
#    Responsibilities:
#      - detect presence of date/time values
#      - classify date vs datetime
#      - resolve AmbiguousDateTime
#      - compute confidence
#      - generate warnings/status metadata
#
#    MUST NOT:
#      - parse strings
#      - normalize values
#      - perform timezone conversion
#      - validate CORE invariants
#      - transform datetime objects
# ============================================================================

from __future__ import annotations

from typing import Any, Dict, Iterable, List, Optional
import datetime

from data_preprocessing.core.date_detector.date_detector_core import DateDetectorCore
from data_preprocessing.core.diagnostics.date_observability import DateObservability
from data_preprocessing.core.contracts.ambiguity_contract import AmbiguousDateTime


class DateDetector:
    """
    DETECTOR LAYER (PFP v4.1.3 STABLE)
    """

    MIN_SAMPLE_SIZE = 10

    def __init__(
        self,
        debug: bool = False,
        strict: bool = False,
        seed: Optional[int] = None,
    ) -> None:

        self.strict = strict

        self._core = DateDetectorCore(
            strict=strict,
            seed=seed,
        )

        self._diag = (
            DateObservability()
            if debug
            else None
        )

    # ===================================================================== #
    # PUBLIC API                                                            #
    # ===================================================================== #

    def detect_dates(
        self,
        values: Iterable[Any],
    ) -> Dict[str, Any]:

        sampled = self._sample(values)

        parsed: List[datetime.datetime] = []

        total = 0
        matches = 0

        for value in sampled:

            if self._is_na(value):
                continue

            total += 1

            resolved = self._core.ensure_datetime(
                self._resolve(value)
            )

            if resolved is not None:
                resolved = self._core.ensure_utc(
                    resolved
                )

            if resolved is None:

                self._emit(
                    value,
                    "REJECTED",
                    "unresolved",
                )

                continue

            parsed.append(resolved)
            matches += 1

            self._emit(
                value,
                "PARSED",
                "ok",
            )

        warnings = self._warnings(
            sampled,
            total,
        )

        if total == 0:

            return {
                "detected": False,
                "date_type": None,
                "confidence": 0.0,
                "normalized_values": [],
                "warnings": warnings,
                "errors": [],
            }

        return {
            "detected": matches > 0,
            "date_type": (
                self._infer_type(parsed)
                if matches > 0
                else "unknown"
            ),
            "confidence": self._confidence(
                matches,
                total,
            ),
            "normalized_values": [],
            "warnings": warnings,
            "errors": [],
        }

    # ===================================================================== #
    # SAMPLING                                                              #
    # ===================================================================== #

    def _sample(
        self,
        values: Iterable[Any],
    ) -> List[Any]:

        return self._core.sample_values(
            list(values)
        )

    # ===================================================================== #
    # NULL HANDLING                                                         #
    # ===================================================================== #

    def _is_na(
        self,
        value: Any,
    ) -> bool:

        return value is None or value == ""

    # ===================================================================== #
    # RESOLUTION                                                            #
    # ===================================================================== #

    def _resolve(
        self,
        value: Any,
    ) -> Optional[datetime.datetime]:

        if isinstance(value, AmbiguousDateTime):

            if not value.candidates:
                return None

            return sorted(
                value.candidates,
                key=lambda c: c.value.isoformat()
            )[0].value

        if isinstance(
            value,
            datetime.datetime,
        ):
            return value

        return None

    # ===================================================================== #
    # CLASSIFICATION                                                        #
    # ===================================================================== #

    def _infer_type(
        self,
        values: List[datetime.datetime],
    ) -> str:

        for value in values:

            if (
                value.hour,
                value.minute,
                value.second,
                value.microsecond,
            ) != (0, 0, 0, 0):

                return "datetime"

        return "date"

    # ===================================================================== #
    # CONFIDENCE                                                            #
    # ===================================================================== #

    def _confidence(
        self,
        matches: int,
        total: int,
    ) -> float:

        if total == 0:
            return 0.0

        rate = matches / total

        coverage = (
            1 -
            1 /
            (1 + total / 20)
        )

        return max(
            0.0,
            min(
                1.0,
                rate * coverage,
            ),
        )

    # ===================================================================== #
    # WARNINGS                                                              #
    # ===================================================================== #

    def _warnings(
        self,
        sampled: List[Any],
        total: int,
    ) -> List[str]:

        if len(sampled) == 0:
            return [
                "no_data"
            ]

        if total < self.MIN_SAMPLE_SIZE:
            return [
                "low_sample_size"
            ]

        return []

    # ===================================================================== #
    # OBSERVABILITY                                                         #
    # ===================================================================== #

    def _emit(
        self,
        value: Any,
        status: str,
        reason: str,
    ) -> None:

        if self._diag:

            self._diag.emit(
                value=value,
                status=status,
                reason=reason,
                meta={
                    "stage": "detector",
                    "strict": self.strict,
                    "spec": "PFP-v4.1.3-STABLE",
                },
            )

from __future__ import annotations
from typing import Any, List, Optional
import datetime

class DateDetectorCore:
    """
    CORE LAYER — PURE INVARIANT ENGINE

    HARD CONTRACT:
    - accepts ONLY datetime.datetime or None (no coercion)
    - rejects all other types → returns None
    - enforces deterministic UTC normalization
    - no parsing, no inference, no enrichment

    SEMANTIC CONTRACT LOCK:
    - CORE is *semantically ignorant* (no ambiguity types, no detector coupling)
    """

    MAX_SAMPLE_SIZE = 500

    # --- GOVERNANCE POLICIES (EXPLICIT SEMANTICS) ---
    TIMEZONE_POLICY = "UTC_ONLY"
    NAIVE_POLICY = "ASSUME_UTC"
    NULL_POLICY = "PROPAGATE"

    # ------------------------------------------------ #
    # INITIALIZATION
    # ------------------------------------------------ #

    def __init__(self, strict: bool = False, seed: Optional[int] = None) -> None:
        self.strict = strict
        self.seed = seed

        if self.TIMEZONE_POLICY != "UTC_ONLY":
            raise ValueError(
                "Unsupported TIMEZONE_POLICY"
            )

        if self.NULL_POLICY not in {
            "PROPAGATE",
            "REJECT",
        }:
            raise ValueError(
                "Unsupported NULL_POLICY"
            )

        if self.NAIVE_POLICY != "ASSUME_UTC":
            raise ValueError(
                "Unsupported NAIVE_POLICY"
            )

    # ------------------------------------------------ #
    # NaN / NULL FIREWALL
    # ------------------------------------------------ #

    def is_na(self, v: Any) -> bool:
        if v is None:
            return True
        try:
            return v != v
        except Exception:
            return False

    # ------------------------------------------------ #
    # SAMPLING (deterministic semantics preserved)
    # ------------------------------------------------ #

    def sample_values(self, values: Any) -> List[Any]:

        if not isinstance(
            values,
            (list, tuple)
        ):
            values = list(values)

        if len(values) <= self.MAX_SAMPLE_SIZE:
            return list(values)

        if self.strict:

            return list(
                values[: self.MAX_SAMPLE_SIZE]
            )

        return list(
            values[: self.MAX_SAMPLE_SIZE]
        )
    # ------------------------------------------------ #
    # TYPE FIREWALL (NO COERCION, NO SEMANTICS)
    # ------------------------------------------------ #

    def ensure_datetime(
        self,
        v: Any
    ) -> Optional[datetime.datetime]:

        if not isinstance(
            v,
            datetime.datetime
        ):
            return None

        return v

    # ------------------------------------------------ #
    # UTC NORMALIZATION (POLICY-GOVERNED)
    # ------------------------------------------------ #

    def ensure_utc(self, dt: Optional[datetime.datetime]) -> Optional[datetime.datetime]:
        if dt is None:
            return None

        if dt.tzinfo is None:
            if self.NAIVE_POLICY == "ASSUME_UTC":
                return dt.replace(tzinfo=datetime.timezone.utc)
            raise ValueError("Naive datetime not allowed under current policy")

        return dt.astimezone(datetime.timezone.utc)

    # ------------------------------------------------ #
    # PRIMARY ENTRYPOINT
    # ------------------------------------------------ #

    def ensure_utc_safe(
        self,
        v: Any
    ) -> Optional[datetime.datetime]:

        dt = self.ensure_datetime(v)

        if dt is None:
            return None

        return self.ensure_utc(dt)

        self,
        max_trace: int = 10_000,

            not isinstance(max_trace, int)
            or max_trace <= 0
        ):
            raise ValueError(
                "max_trace must be positive integer"
            )

            maxlen=max_trace
        )


    def emit(
        self,
        value: Any,
        status: str,
        reason: Optional[str] = None,
        meta: Optional[Dict[str, Any]] = None,
    ) -> None:
        """
        Append immutable snapshot.

        No validation.
        No interpretation.
        """

        try:
            value_snapshot = copy.deepcopy(value)
        except Exception:
            value_snapshot = "<unavailable>"


        try:
            meta_snapshot = copy.deepcopy(
                meta or {}
            )
        except Exception:
            meta_snapshot = {}


        self._trace.append(
            Event(
                value=value_snapshot,
                status=status,
                reason=reason,
                meta=meta_snapshot,
            )
        )


    def get_trace(self) -> List[Event]:
        """
        Returns detached snapshot.
        """

        return list(self._trace)


    def clear(self) -> None:
        """
        Testing/replay lifecycle only.
        """

        self._trace.clear()


    def size(self) -> int:
        return len(self._trace)

@dataclass(frozen=True)
class MetricsSnapshot:
    parsed: int = 0
    rejected: int = 0
    skipped: int = 0

    def success_rate(self) -> float:
        total = (
            self.parsed
            + self.rejected
            + self.skipped
        )

        if total == 0:
            return 0.0

        return self.parsed / total



# ============================================================================
# METRICS EXPORTER
# ============================================================================

class MetricsExporter:

    STATUS_PARSED = "PARSED"
    STATUS_REJECTED = "REJECTED"
    STATUS_SKIPPED = "SKIPPED"


    def __init__(
        self,
        max_trace: int = 10_000,
        sample_rate: float = 0.1,
        deterministic_seed: Optional[int] = None,
    ) -> None:

        if (
            not isinstance(max_trace, int)
            or max_trace <= 0
        ):
            raise ValueError(
                "max_trace must be positive integer"
            )


        if not 0.0 <= sample_rate <= 1.0:
            raise ValueError(
                "sample_rate must be between 0 and 1"
            )


        self._max_trace = max_trace
        self._sample_rate = sample_rate
        self._seed = deterministic_seed


        self._trace = deque(
            maxlen=max_trace
        )


        self._rng = (
            random.Random(deterministic_seed)
            if deterministic_seed is not None
            else random.SystemRandom()
        )


        self._metrics = {
            self.STATUS_PARSED: 0,
            self.STATUS_REJECTED: 0,
            self.STATUS_SKIPPED: 0,
        }


    # ----------------------------------------------------------------------
    # EMIT
    # ----------------------------------------------------------------------

    def emit(
        self,
        value: Any,
        status: str,
        reason: str = "",
    ) -> None:

        """
        Diagnostics emission.

        Never raises.
        Never affects caller.
        """

        try:

            if status not in self._metrics:
                return


            self._metrics[status] = (
                self._metrics.get(
                    status,
                    0
                ) + 1
            )


            if not self._should_sample():
                return


            self._trace.append(
                {
                    "ts": time.time(),
                    "value_type": type(value).__name__,
                    "value": self._safe_repr(value),
                    "status": status,
                    "reason": str(reason)[:200],
                }
            )


        except Exception:
            # Diagnostics isolation rule.
            # Never leak observability failures.
            return



    # ----------------------------------------------------------------------
    # SAMPLING
    # ----------------------------------------------------------------------

    def _should_sample(self) -> bool:

        if self._sample_rate >= 1.0:
            return True

        if self._sample_rate <= 0.0:
            return False


        if self._seed is not None:

            return (
                self._rng.random()
                <
                self._sample_rate
            )


        return (
            self._rng.random()
            <
            self._sample_rate
        )



    # ----------------------------------------------------------------------
    # SAFE SERIALIZATION
    # ----------------------------------------------------------------------

    def _safe_repr(
        self,
        value: Any,
    ) -> str:

        try:
            return repr(value)[:200]

        except Exception:
            return "<unrepresentable>"



    # ----------------------------------------------------------------------
    # READ API
    # ----------------------------------------------------------------------

    def get_trace(
        self,
    ) -> List[Dict[str, Any]]:

        return copy.deepcopy(
            list(self._trace)
        )


    def get_metrics(
        self,
    ) -> MetricsSnapshot:

        return MetricsSnapshot(
            parsed=self._metrics[
                self.STATUS_PARSED
            ],
            rejected=self._metrics[
                self.STATUS_REJECTED
            ],
            skipped=self._metrics[
                self.STATUS_SKIPPED
            ],
        )



    # ----------------------------------------------------------------------
    # RESET
    # ----------------------------------------------------------------------

    def reset(
        self,
    ) -> None:

        self._trace.clear()

        self._metrics = {
            self.STATUS_PARSED: 0,
            self.STATUS_REJECTED: 0,
            self.STATUS_SKIPPED: 0,
        }


        if self._seed is not None:
            self._rng = random.Random(
                self._seed
            )

class FingerprintCanonicalizationError(TypeError):
    """
    Controlled failure for invalid fingerprint payloads.

    Contract guarantees:
    - deterministic failure type
    - no leaked serializer exceptions
    - safe handling by callers
    """

    pass

def _json_sort_key(value: Any) -> str:
    """
    Generates deterministic ordering key for unordered collections.
    """

    return json.dumps(
        value,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
        allow_nan=False,
    )


# ============================================================
#  CANONICALIZATION
# ============================================================

def _canonicalize(
    value: Any,
    *,
    _seen: set[int] | None = None,
) -> Any:
    """
    HQ2 STRICT canonicalization engine.

    Guarantees:

    Containers:
    - dict keys MUST be strings
    - tuple/list normalize to JSON arrays
    - set/frozenset normalize deterministically

    Scalars:
    - bool preserved
    - Enum serialized by value
    - NaN rejected
    - Infinity rejected

    Safety:
    - cyclic structures rejected
    - unsupported objects rejected
    """

    if _seen is None:
        _seen = set()

    # --------------------------------------------------------
    # Primitive fast paths
    # --------------------------------------------------------

    if value is None:
        return None

    if isinstance(value, bool):
        return value

    if isinstance(value, Enum):
        return _canonicalize(value.value, _seen=_seen)

    if isinstance(value, (str, int)):
        return value

    if isinstance(value, float):
        if not math.isfinite(value):
            raise FingerprintCanonicalizationError(
                "NaN and Infinity are forbidden in fingerprint input"
            )

        return value

    if isinstance(
        value,
        datetime.datetime
    ):
        raise FingerprintCanonicalizationError(
            "Datetime values require explicit serialization"
        )


    if isinstance(value, (bytes, bytearray, memoryview)):
        raise FingerprintCanonicalizationError(
            "Binary values are not supported in fingerprint input"
        )

    # --------------------------------------------------------
    # Cycle protection
    # --------------------------------------------------------

    object_id = id(value)

    if object_id in _seen:
        raise FingerprintCanonicalizationError(
            "Cyclic references are not allowed in fingerprint input"
        )

    _seen.add(object_id)

    try:

        # ----------------------------------------------------
        # Dictionary
        # ----------------------------------------------------

        if isinstance(value, dict):

            for key in value.keys():

                if not isinstance(key, str):
                    raise FingerprintCanonicalizationError(
                        "Fingerprint dict keys must be strings"
                    )

            return {
                key: _canonicalize(item, _seen=_seen)
                for key, item in sorted(
                    value.items(),
                    key=lambda item: item[0],
                )
            }

        # ----------------------------------------------------
        # Sequence
        # ----------------------------------------------------

        if isinstance(value, (list, tuple)):

            return [
                _canonicalize(item, _seen=_seen)
                for item in value
            ]

        # ----------------------------------------------------
        # Unordered collections
        # ----------------------------------------------------

        if isinstance(value, (set, frozenset)):

            normalized = [
                _canonicalize(item, _seen=_seen)
                for item in value
            ]

            return sorted(
                normalized,
                key=_json_sort_key,
            )

        # ----------------------------------------------------
        # Unsupported objects
        # ----------------------------------------------------

        raise FingerprintCanonicalizationError(
            f"Unsupported fingerprint input type: {type(value).__name__}"
        )

    finally:
        _seen.remove(object_id)


# ============================================================
#  SHA256 GENERATOR
# ============================================================

def generate_sha256(data: Any) -> str:
    """
    Generates deterministic SHA-256 fingerprint.

    Governance contract:

    - same logical structure => same fingerprint
    - dict ordering does not affect result
    - set ordering does not affect result
    - tuple/list are normalized identically
    - Unicode preserved
    - NaN/Infinity forbidden
    - cyclic structures forbidden

    Returns:
        64-character lowercase hexadecimal SHA256 digest.

    Domain rules MUST NOT exist here.
    """

    canonical = _canonicalize(data)

    serialized = json.dumps(
        canonical,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
        allow_nan=False,
    ).encode("utf-8")

    digest = hashlib.sha256(serialized).hexdigest()

    return digest

            max_depth_encountered=depth,
        )


    def flatten_records(
        self,
        records: Iterable[Dict[str, Any]],
    ) -> Dict[str, Any]:

        if records is None:
            raise HQ2FlatteningError(
                "records cannot be None"
            )

        flattened_records: List[Dict[str, Any]] = []
        events: List[Dict[str, Any]] = []


        for record in records:

            result = self.flatten_record(record)

            flattened_records.append(
                result["flattened_record"]
            )

            events.extend(
                result["events"]
            )


        return make_flatten_records_result(
            flattened_records=flattened_records,
            events=events,
        )


    # ========================================================
    # INTERNAL ENGINE
    # ========================================================

    def _flatten_dict(
        self,
        data: Dict[str, Any],
        parent_path: str,
        current_depth: int,
        active_ids: Set[int],
        events: List[Dict[str, Any]],
    ) -> Tuple[Dict[str, Any], int]:

        if current_depth > self.max_depth:

            raise HQ2MaxDepthExceededError(
                f"MAX_DEPTH_EXCEEDED:{parent_path}"
            )


        object_id = id(data)

        if object_id in active_ids:

            raise HQ2CircularReferenceError(
                f"CIRCULAR_REFERENCE:{parent_path}"
            )


        active_ids.add(object_id)

        result: Dict[str, Any] = {}

        max_depth = current_depth


        try:

            for raw_key in sorted(
                data.keys(),
                key=str,
            ):

                value = data[raw_key]

                key = self._sanitize_key(
                    str(raw_key)
                )


                path = (
                    key
                    if not parent_path
                    else
                    f"{parent_path}{self.separator}{key}"
                )


                if isinstance(value, dict):

                    if not value:

                        self._insert(
                            result,
                            path,
                            {},
                            events,
                        )

                    else:

                        nested, nested_depth = self._flatten_dict(
                            value,
                            path,
                            current_depth + 1,
                            active_ids,
                            events,
                        )

                        max_depth = max(
                            max_depth,
                            nested_depth,
                        )

                        for (
                            nested_key,
                            nested_value,
                        ) in nested.items():

                            self._insert(
                                result,
                                nested_key,
                                nested_value,
                                events,
                            )


                else:

                    self._insert(
                        result,
                        path,
                        value,
                        events,
                    )


        finally:

            active_ids.discard(object_id)


        return result, max_depth


    # ========================================================
    # HELPERS
    # ========================================================

    def _insert(
        self,
        target: Dict[str, Any],
        key: str,
        value: Any,
        events: List[Dict[str, Any]],
    ) -> None:


        if key not in target:

            target[key] = value
            return


        index = 1

        while (
            f"{key}_collision_{index}"
            in target
        ):

            index += 1


        replacement = (
            f"{key}_collision_{index}"
        )


        events.append(
            {
                "type": "KEY_COLLISION",
                "source": key,
                "target": replacement,
            }
        )


        target[replacement] = value



    def _sanitize_key(
        self,
        key: str,
    ) -> str:

        return key.replace(
            self.separator,
            "_",
        )
