from dataclasses import dataclass
from typing import Final, Sequence

    "VERSION",
    "DetectionMultivariateAnomalyContractError",
    "DetectionMultivariateAnomalyContract",
]

VERSION: Final[str] = "DetectionMultivariateAnomalyContract v6.3"


class DetectionMultivariateAnomalyContractError(Exception):
    pass


@dataclass(frozen=True, slots=True)
class DetectionMultivariateAnomalyContract:
    record_id: str
    involved_features: Sequence[str]
    anomaly_score: float
    model_name: str  # e.g., 'ISOLATION_FOREST', 'AUTOENCODER'
    version: str = VERSION

    def __post_init__(self) -> None:
        if not isinstance(self.record_id, str):
            raise DetectionMultivariateAnomalyContractError("RECORD_ID_MUST_BE_STRING")
        if not self.record_id.strip():
            raise DetectionMultivariateAnomalyContractError("RECORD_ID_EMPTY")

        if not isinstance(self.involved_features, Sequence):
            raise DetectionMultivariateAnomalyContractError("INVOLVED_FEATURES_MUST_BE_SEQUENCE")
        if not self.involved_features:
            raise DetectionMultivariateAnomalyContractError("INVOLVED_FEATURES_CANNOT_BE_EMPTY")
        for feature in self.involved_features:
            if not isinstance(feature, str):
                raise DetectionMultivariateAnomalyContractError("INVOLVED_FEATURES_MUST_CONTAIN_STRINGS")

        if not isinstance(self.anomaly_score, float):
            raise DetectionMultivariateAnomalyContractError("ANOMALY_SCORE_MUST_BE_FLOAT")
        if not (0.0 <= self.anomaly_score <= 1.0):
            raise DetectionMultivariateAnomalyContractError("ANOMALY_SCORE_MUST_BE_BETWEEN_0_AND_1")

        if not isinstance(self.model_name, str):
            raise DetectionMultivariateAnomalyContractError("MODEL_NAME_MUST_BE_STRING")
        if not self.model_name.strip():
            raise DetectionMultivariateAnomalyContractError("MODEL_NAME_EMPTY")

        if not isinstance(self.version, str):
            raise DetectionMultivariateAnomalyContractError("VERSION_MUST_BE_STRING")
        if not self.version.strip():
            raise DetectionMultivariateAnomalyContractError("VERSION_EMPTY")

@dataclass(frozen=True, slots=True)
class DetectionOutlierBoundContract:
    method_name: str  # e.g., 'Z-SCORE', 'IQR', 'ISOLATION_FOREST'
    lower_bound: float | None
    upper_bound: float | None
    outlier_ratio: float
    version: str = VERSION

    def __post_init__(self) -> None:
        if not isinstance(self.method_name, str):
            raise DetectionOutlierBoundContractError("METHOD_NAME_MUST_BE_STRING")
        if not self.method_name.strip():
            raise DetectionOutlierBoundContractError("METHOD_NAME_EMPTY")

        if self.lower_bound is not None and not isinstance(self.lower_bound, float):
            raise DetectionOutlierBoundContractError("LOWER_BOUND_MUST_BE_FLOAT_OR_NONE")

        if self.upper_bound is not None and not isinstance(self.upper_bound, float):
            raise DetectionOutlierBoundContractError("UPPER_BOUND_MUST_BE_FLOAT_OR_NONE")

        if self.lower_bound is not None and self.upper_bound is not None:
            if self.lower_bound > self.upper_bound:
                raise DetectionOutlierBoundContractError("LOWER_BOUND_CANNOT_EXCEED_UPPER_BOUND")

        if not isinstance(self.outlier_ratio, float):
            raise DetectionOutlierBoundContractError("OUTLIER_RATIO_MUST_BE_FLOAT")
        if not (0.0 <= self.outlier_ratio <= 1.0):
            raise DetectionOutlierBoundContractError("OUTLIER_RATIO_MUST_BE_BETWEEN_0_AND_1")

        if not isinstance(self.version, str):
            raise DetectionOutlierBoundContractError("VERSION_MUST_BE_STRING")
        if not self.version.strip():
            raise DetectionOutlierBoundContractError("VERSION_EMPTY")

            raise DetectionSparsityContractError("MISSING_RECORDS_EXCEED_TOTAL_RECORDS")

                raise IngestionPartitionRoutingContractError("PARTITION_KEYS_MUST_CONTAIN_STRINGS")

        if not isinstance(self.partition_values, Sequence):
            raise IngestionPartitionRoutingContractError("PARTITION_VALUES_MUST_BE_SEQUENCE")
            
        for val in self.partition_values:
            if not isinstance(val, str) or not val.strip():
                raise IngestionPartitionRoutingContractError("PARTITION_VALUES_MUST_CONTAIN_STRINGS")

        if len(self.partition_keys) != len(self.partition_values):
            raise IngestionPartitionRoutingContractError("PARTITION_KEYS_AND_VALUES_LENGTH_MISMATCH")

        if not isinstance(self.version, str):
            raise IngestionPartitionRoutingContractError("VERSION_MUST_BE_STRING")
        if not self.version.strip():
            raise IngestionPartitionRoutingContractError("VERSION_EMPTY")
