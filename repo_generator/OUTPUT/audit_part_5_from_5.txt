@dataclass(frozen=True, slots=True)

    previous_node_count: int
    new_node_count: int
    version: str = VERSION

    def __post_init__(self) -> None:
        if not isinstance(self.cluster_id, str):
            raise SystemClusterScalingEventContractError("CLUSTER_ID_MUST_BE_STRING")
        if not self.cluster_id.strip():
            raise SystemClusterScalingEventContractError("CLUSTER_ID_EMPTY")

        if not isinstance(self.scaling_action, str):
            raise SystemClusterScalingEventContractError("SCALING_ACTION_MUST_BE_STRING")
        if self.scaling_action not in ("SCALE_OUT", "SCALE_IN"):
            raise SystemClusterScalingEventContractError("SCALING_ACTION_MUST_BE_VALID_ENUM")

        if not isinstance(self.previous_node_count, int):
            raise SystemClusterScalingEventContractError("PREVIOUS_NODE_COUNT_MUST_BE_INT")
        if self.previous_node_count < 0:

from __future__ import annotations

from dataclasses import dataclass
from typing import Final


__all__ = [
    "VERSION",
    "SystemZombieProcessReaperContractError",
    "SystemZombieProcessReaperContract",
]

VERSION: Final[str] = "SystemZombieProcessReaperContract v6.3"


class SystemZombieProcessReaperContractError(Exception):
    pass


@dataclass(frozen=True, slots=True)
class SystemZombieProcessReaperContract:
    node_id: str
    reaped_zombie_pid: int
    parent_pid: int
    is_parent_killed: bool
    version: str = VERSION

    def __post_init__(self) -> None:
        if not isinstance(self.node_id, str):
            raise SystemZombieProcessReaperContractError("NODE_ID_MUST_BE_STRING")
        if not self.node_id.strip():
            raise SystemZombieProcessReaperContractError("NODE_ID_EMPTY")

        if not isinstance(self.reaped_zombie_pid, int):
            raise SystemZombieProcessReaperContractError("REAPED_ZOMBIE_PID_MUST_BE_INT")
        if self.reaped_zombie_pid <= 0:
            raise SystemZombieProcessReaperContractError("REAPED_ZOMBIE_PID_MUST_BE_POSITIVE")

        if not isinstance(self.parent_pid, int):
            raise SystemZombieProcessReaperContractError("PARENT_PID_MUST_BE_INT")
        if self.parent_pid <= 0:
            raise SystemZombieProcessReaperContractError("PARENT_PID_MUST_BE_POSITIVE")

        if not isinstance(self.is_parent_killed, bool):
            raise SystemZombieProcessReaperContractError("IS_PARENT_KILLED_MUST_BE_BOOL")

        if not isinstance(self.version, str):
            raise SystemZombieProcessReaperContractError("VERSION_MUST_BE_STRING")
        if not self.version.strip():
            raise SystemZombieProcessReaperContractError("VERSION_EMPTY")
