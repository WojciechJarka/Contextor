class AdapterAuditError(Exception):
    """
    Adapter audit structural artifact failure.

    Represents:

        invalid contract structure only.

    Does not represent:

        adapter failures,
        conversion failures,
        runtime failures.
    """

    pass



# ============================================================
# ADAPTER AUDIT ENTRY
# ============================================================


@dataclass(
    frozen=True,
    slots=True,
)
class AdapterAuditEntry:
    """
    Immutable adapter evidence record.

    Represents:

        observed adapter conversion fact.

    Does not represent:

        semantic correctness,
        business decision,
        recovery action.
    """


    adapter_name: str

    input_type: str

    output_type: str

    status: str

    message: str = ""



    def __post_init__(
        self,
    ) -> None:


        for name, value in (
            ("adapter_name", self.adapter_name),
            ("input_type", self.input_type),
            ("output_type", self.output_type),
            ("status", self.status),
            ("message", self.message),
        ):

            if not isinstance(
                value,
                str,
            ):
                raise AdapterAuditError(
                    f"{name.upper()}_MUST_BE_STRING"
                )


            if name != "message":

                if not value.strip():

                    raise AdapterAuditError(
                        f"{name.upper()}_EMPTY"
                    )



# ============================================================
# AUDIT REPORT CONTRACT
# ============================================================


@dataclass(
    frozen=True,
    slots=True,
)
class AdapterReportAudit:
    """
    Immutable adapter audit report.

    Contains:

        - adapter identity
        - source contract reference
        - conversion observations

    Does not:

        - execute adapters
        - validate adapter correctness
        - mutate pipeline state
    """


    version: str

    source_contract_version: str

    adapter: str


    entries: Tuple[
        AdapterAuditEntry,
        ...
    ]



    def __post_init__(
        self,
    ) -> None:


        for name, value in (
            ("version", self.version),
            (
                "source_contract_version",
                self.source_contract_version,
            ),
            ("adapter", self.adapter),
        ):

            if not isinstance(
                value,
                str,
            ):
                raise AdapterAuditError(
                    f"{name.upper()}_MUST_BE_STRING"
                )


            if not value.strip():

                raise AdapterAuditError(
                    f"{name.upper()}_EMPTY"
                )



        if not isinstance(
            self.entries,
            tuple,
        ):

            raise AdapterAuditError(
                "ENTRIES_MUST_BE_TUPLE"
            )



        for entry in self.entries:

            if not isinstance(
                entry,
                AdapterAuditEntry,
            ):

                raise AdapterAuditError(
                    "INVALID_ADAPTER_AUDIT_ENTRY"
                )



# ============================================================
# READ HELPERS
# ============================================================


    @property
    def entry_count(
        self,
    ) -> int:
        """
        Deterministic observation count.
        """

        return len(
            self.entries
        )



    @property
    def passed(
        self,
    ) -> bool:
        """
        Evidence state projection.

        Does not evaluate adapter correctness.
        """

        return all(
            entry.status.upper() == "OK"
            for entry in self.entries
        )

# ============================================================


@dataclass(
    frozen=True,
    slots=True,
)
class AmbiguityCandidateAudit:
    """
    Immutable ambiguity candidate observation.

    Represents possible interpretation
    detected by ingestion layer.

    Does not represent:
        - selected resolution
        - semantic decision
    """

    value: str

    confidence: float



    def __post_init__(
        self,
    ) -> None:


        if not isinstance(
            self.value,
            str,
        ):
            raise AmbiguityAuditError(
                "CANDIDATE_VALUE_MUST_BE_STRING"
            )


        if not self.value.strip():

            raise AmbiguityAuditError(
                "CANDIDATE_VALUE_EMPTY"
            )


        if not isinstance(
            self.confidence,
            (float, int),
        ):

            raise AmbiguityAuditError(
                "CONFIDENCE_MUST_BE_NUMERIC"
            )


        if (
            self.confidence < 0.0
            or self.confidence > 1.0
        ):

            raise AmbiguityAuditError(
                "CONFIDENCE_OUT_OF_RANGE"
            )

# ============================================================
# AUDIT EXECUTOR
# ============================================================


def audit_ingestion_structure(
    report: IngestionStructureReport,
) -> IngestionStructureAuditReport:
    """
    Deterministic structure governance evaluation.

    Input:
        immutable CORE structure facts

    Output:
        immutable AUDIT evidence artifact

    Guarantees:
        - no mutation
        - no repair
        - no inference
    """


    if not isinstance(
        report,
        IngestionStructureReport,
    ):
        raise StructureAuditError(
            "INVALID_STRUCTURE_REPORT_SOURCE"
        )


    issues: list[StructureAuditIssue] = []


    # --------------------------------------------------------------------- #
    # REPORT LEVEL RULES                                                    #
    # --------------------------------------------------------------------- #


    if report.total_nodes < 0:

        issues.append(
            StructureAuditIssue(
                rule_id=(
                    "STRUCTURE.REPORT."
                    "TOTAL_NODES.NON_NEGATIVE"
                ),
                severity="ERROR",
                message=(
                    "total_nodes cannot be negative"
                ),
            )
        )


    if report.max_depth < 0:

        issues.append(
            StructureAuditIssue(
                rule_id=(
                    "STRUCTURE.REPORT."
                    "MAX_DEPTH.NON_NEGATIVE"
                ),
                severity="ERROR",
                message=(
                    "max_depth cannot be negative"
                ),
            )
        )


    # --------------------------------------------------------------------- #
    # NODE LEVEL RULES                                                      #
    # --------------------------------------------------------------------- #


    for node in report.nodes:


        if node.depth < 0:

            issues.append(
                StructureAuditIssue(
                    rule_id=(
                        "STRUCTURE.NODE."
                        "DEPTH.NON_NEGATIVE"
                    ),
                    severity="ERROR",
                    message=(
                        "node depth cannot be negative"
                    ),
                    path=node.path,
                )
            )


        if node.children_count < 0:

            issues.append(
                StructureAuditIssue(
                    rule_id=(
                        "STRUCTURE.NODE."
                        "CHILDREN.NON_NEGATIVE"
                    ),
                    severity="ERROR",
                    message=(
                        "children_count cannot be negative"
                    ),
                    path=node.path,
                )
            )


        if not node.path:

            issues.append(
                StructureAuditIssue(
                    rule_id=(
                        "STRUCTURE.NODE."
                        "PATH.REQUIRED"
                    ),
                    severity="ERROR",
                    message=(
                        "node path cannot be empty"
                    ),
                )
            )


    return IngestionStructureAuditReport(
        audit_version=VERSION,
        source_contract_version=report.version,
        valid=(
            len(issues) == 0
        ),
        issues=tuple(
            issues
        ),
    )

# ============================================================
# IMMUTABILITY HELPERS

def _validate_string(
    name: str,
    value: Any,
) -> None:

    if not isinstance(
        value,
        str,
    ):
        raise NormalizationAuditError(
            f"{name.upper()}_MUST_BE_STRING"
        )


    if not value.strip():

        raise NormalizationAuditError(
            f"{name.upper()}_EMPTY"
        )

# ============================================================


@dataclass(
    frozen=True,
    slots=True,
)
class NormalizationObservation:
    """
    Immutable normalization evidence record.

    Represents:

        observed normalization event.

    Does not represent:

        normalization execution,
        normalization policy,
        semantic correctness.
    """


    field: str

    operation: str

    status: str

    original_type: str | None = None

    normalized_type: str | None = None



    def __post_init__(
        self,
    ) -> None:


        for name, value in (
            (
                "field",
                self.field,
            ),
            (
                "operation",
                self.operation,
            ),
            (
                "status",
                self.status,
            ),
        ):

            _validate_string(
                name,
                value,
            )


        for name, value in (
            (
                "original_type",
                self.original_type,
            ),
            (
                "normalized_type",
                self.normalized_type,
            ),
        ):

            if value is not None:

                _validate_string(
                    name,
                    value,
                )

# REPORT CONTRACT
# ============================================================


@dataclass(
    frozen=True,
    slots=True,
)
class NormalizationReportAudit:
    """
    Immutable normalization audit artifact.

    Contains:

        - normalization observations
        - evidence metadata
        - source contract identity

    Does not:

        - execute normalization
        - decide normalization rules
        - validate business correctness
    """


    source_contract_version: str

    normalizer_name: str


    observations: Tuple[
        NormalizationObservation,
        ...
    ]


    metadata: Mapping[
        str,
        Any,
    ] | None = None


    version: str = VERSION



    def __post_init__(
        self,
    ) -> None:


        _validate_string(
            "source_contract_version",
            self.source_contract_version,
        )


        _validate_string(
            "normalizer_name",
            self.normalizer_name,
        )


        if not isinstance(
            self.observations,
            tuple,
        ):
            raise NormalizationAuditError(
                "OBSERVATIONS_MUST_BE_TUPLE"
            )


        for observation in self.observations:

            if not isinstance(
                observation,
                NormalizationObservation,
            ):
                raise NormalizationAuditError(
                    "INVALID_NORMALIZATION_OBSERVATION"
                )


        if self.metadata is not None:

            if not isinstance(
                self.metadata,
                Mapping,
            ):
                raise NormalizationAuditError(
                    "METADATA_MUST_BE_MAPPING"
                )


        _validate_string(
            "version",
            self.version,
        )



# ============================================================
# READ HELPERS
# ============================================================


    @property
    def observation_count(
        self,
    ) -> int:
        """
        Deterministic observation count.
        """

        return len(
            self.observations
        )



    @property
    def passed(
        self,
    ) -> bool:
        """
        Evidence completion state.

        Does not evaluate semantic correctness.
        """

        return all(
            item.status.upper() == "OK"
            for item in self.observations
        )



    def fields(
        self,
    ) -> Tuple[str, ...]:
        """
        Deterministic normalized field identity list.
        """

        return tuple(
            item.field
            for item in self.observations
        )



    def operations(
        self,
    ) -> Tuple[str, ...]:
        """
        Deterministic normalization operation list.
        """

        return tuple(
            item.operation
            for item in self.observations
        )

@dataclass(

    """

        ingestion field and ontology identifier.

    Does not represent:

        ontology resolution,
        semantic correctness,
        business interpretation.
    """


    field: str

    ontology_id: str

    status: str

    confidence: float | None = None

@dataclass(
    frozen=True,
    slots=True,
)
class ValidationClosure:
    """
    Immutable validation lifecycle closure artifact.

    Contains:

        - validation events
        - validator identity
        - pipeline identity
        - closure state

    Does not:

        - execute validation
        - evaluate remediation
        - modify runtime state
    """

    pipeline: str

    validator: str

    events: Tuple[
        ValidationEvent,
        ...
    ]

    closed: bool

    version: str = VERSION



    def __post_init__(
        self,
    ) -> None:


        for name, value in (
            (
                "pipeline",
                self.pipeline,
            ),
            (
                "validator",
                self.validator,
            ),
        ):

            if not isinstance(
                value,
                str,
            ):

                raise ValidationClosureError(
                    f"{name.upper()}_MUST_BE_STRING"
                )


            if not value.strip():

                raise ValidationClosureError(
                    f"{name.upper()}_EMPTY"
                )



        if not isinstance(
            self.events,
            tuple,
        ):

            raise ValidationClosureError(
                "EVENTS_MUST_BE_TUPLE"
            )



        for event in self.events:

            if not isinstance(
                event,
                ValidationEvent,
            ):

                raise ValidationClosureError(
                    "INVALID_VALIDATION_EVENT"
                )



        if not isinstance(
            self.closed,
            bool,

            raise ValidationClosureError(

        Does not execute validation.

            and all(
                event.status.upper() == "OK"
                for event in self.events
            )
        )
