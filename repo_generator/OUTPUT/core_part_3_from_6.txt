@dataclass(frozen=True, slots=True)
class PipelineResultContract:
    run_info: PipelineRunContract
    final_status: str  # e.g., 'SUCCESS', 'FAILED'
    tasks_succeeded_count: int
    tasks_failed_count: int
    diagnostics: Sequence[PipelineDiagnosticContract]
    version: str = VERSION

    def __post_init__(self) -> None:
        if not isinstance(self.run_info, PipelineRunContract):
            raise PipelineResultContractError("RUN_INFO_MUST_BE_PIPELINE_RUN_CONTRACT")

        if not isinstance(self.final_status, str):
            raise PipelineResultContractError("FINAL_STATUS_MUST_BE_STRING")
        if self.final_status not in ("SUCCESS", "FAILED", "KILLED"):
            raise PipelineResultContractError("FINAL_STATUS_MUST_BE_VALID_ENUM")

        if not isinstance(self.tasks_succeeded_count, int):
            raise PipelineResultContractError("TASKS_SUCCEEDED_COUNT_MUST_BE_INT")
        if self.tasks_succeeded_count < 0:
            raise PipelineResultContractError("TASKS_SUCCEEDED_COUNT_MUST_BE_NON_NEGATIVE")

        if not isinstance(self.tasks_failed_count, int):
            raise PipelineResultContractError("TASKS_FAILED_COUNT_MUST_BE_INT")
        if self.tasks_failed_count < 0:
            raise PipelineResultContractError("TASKS_FAILED_COUNT_MUST_BE_NON_NEGATIVE")

        if not isinstance(self.diagnostics, Sequence):
            raise PipelineResultContractError("DIAGNOSTICS_MUST_BE_SEQUENCE")
        for diag in self.diagnostics:
            if not isinstance(diag, PipelineDiagnosticContract):
                raise PipelineResultContractError("DIAGNOSTICS_MUST_CONTAIN_CONTRACTS")

        if not isinstance(self.version, str):
            raise PipelineResultContractError("VERSION_MUST_BE_STRING")
        if not self.version.strip():
            raise PipelineResultContractError("VERSION_EMPTY")

@dataclass(frozen=True, slots=True)
class PipelineScheduleContract:
    cron_expression: str | None
    interval_seconds: int | None
    catchup_enabled: bool
    version: str = VERSION

    def __post_init__(self) -> None:
        if self.cron_expression is None and self.interval_seconds is None:
            raise PipelineScheduleContractError("MUST_PROVIDE_CRON_OR_INTERVAL")
        
        if self.cron_expression is not None and self.interval_seconds is not None:
            raise PipelineScheduleContractError("CANNOT_PROVIDE_BOTH_CRON_AND_INTERVAL")

        if self.cron_expression is not None and not isinstance(self.cron_expression, str):
            raise PipelineScheduleContractError("CRON_EXPRESSION_MUST_BE_STRING_OR_NONE")

        if self.interval_seconds is not None:
            if not isinstance(self.interval_seconds, int):
                raise PipelineScheduleContractError("INTERVAL_SECONDS_MUST_BE_INT_OR_NONE")
            if self.interval_seconds <= 0:
                raise PipelineScheduleContractError("INTERVAL_SECONDS_MUST_BE_POSITIVE")

        if not isinstance(self.catchup_enabled, bool):
            raise PipelineScheduleContractError("CATCHUP_ENABLED_MUST_BE_BOOL")

        if not isinstance(self.version, str):
            raise PipelineScheduleContractError("VERSION_MUST_BE_STRING")
        if not self.version.strip():
            raise PipelineScheduleContractError("VERSION_EMPTY")
